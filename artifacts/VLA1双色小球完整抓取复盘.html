  31  def parse_compound_instruction(text: str) -> list[PlanStep]:
  32      normalized = text.strip().lower().replace("的球", "球")
  33      matches = list(re.finditer(
  34          r"把\s*(红色|绿色|红|绿)球?\s*放到\s*(红色|绿色|红|绿)"
  35          r"(?:球)?\s*(旁边|正前方|前方|正前面|前面|原来的位置|原位置)", normalized
  36      ))
  37      steps = []
  38      for match in matches:
  39          relation = {
  40              "旁边": "beside",
  41              "正前方": "front",
  42              "前方": "front",
  43              "正前面": "front",
  44              "前面": "front",
  45              "原来的位置": "original",
  46              "原位置": "original",
  47          }[match.group(3)]
  48          steps.append(PlanStep(
  49              COLORS_ZH[match.group(1)], relation, COLORS_ZH[match.group(2)],
  50              match.group(0),
  51          ))
  52      if not steps:
  53          english = re.finditer(
  54              r"(?:place|put|move)\s+(?:the\s+)?(red|green)\s+(?:ball\s+)?"
  55              r"(next to|beside|in front of|at)\s+(?:the\s+)?(red|green)(?:'s)?\s*"
  56              r"(original (?:position|location))?", normalized
  57          )
  58          for match in english:
  59              relation = (
  60                  "original" if match.group(4)
  61                  else "front" if match.group(2) == "in front of"
  62                  else "beside"
  63              )
  64              steps.append(PlanStep(
  65                  match.group(1), relation, match.group(3), match.group(0),
  66              ))
  67      if not steps:
  68          raise ValueError("无法解析任务。请使用“把绿色的球放到红色旁边”这样的表达。")
  69      expected = len(re.findall(
  70          r"把\s*(?:红色|绿色|红|绿).*?放到", normalized
  71      ))
  72      expected += len(re.findall(r"\b(?:place|put|move)\b", normalized))
  73      if expected != len(steps):
  74          raise ValueError(
  75              f"指令包含 {expected} 个动作，但只识别出 {len(steps)} 个；"
  76              "系统已停止，避免只执行部分任务。"
  77          )
  78      for step in steps:
  79          if step.target == step.reference:
  80              raise ValueError("目标球和参考球不能是同一个颜色。")
  81      return steps

 162              results.append(StepResult(step, np.zeros(3), False, False,
 163                                        "vision_failed", 0))
 164              break
 165          reference = tracker.require(step.reference)
 166          future_destination = None

 168              next_step = plan[index]

 172              goal = safe_beside_goal(reference, future_destination)

 185              open_action=-1.0, open_steps=18, transport_height=0.24,
 186          )
 187          completed = phase == "done"
 188          results.append(StepResult(step, goal, completed, bool(success),

  36  def pixel_to_table(pixel, matrix, z=BALL_CENTER_Z):
  37      # For fixed z, the 3D projection becomes a 3x3 planar homography.
  38      homography = np.column_stack(
  39          (matrix[:3, 0], matrix[:3, 1], z * matrix[:3, 2] + matrix[:3, 3])
  40      )
  41      row, column = pixel
  42      # This project uses robosuite's OpenGL image convention (bottom row first),
  43      # while the pinhole matrix follows OpenCV's top-row-first convention.
  44      row = IMAGE_SIZE - 1 - row
  45      world_plane = np.linalg.solve(homography, np.array([column, row, 1.0]))
  46      world_plane /= world_plane[2]
  47      return np.array([world_plane[0], world_plane[1], z], dtype=np.float64)

  49  def color_mask(image, color):
  50      rgb = image.astype(np.int16)
  51      red, green, blue = rgb[..., 0], rgb[..., 1], rgb[..., 2]
  52      if color == "red":
  53          return (red > 90) & (red > green * 1.35) & (red > blue * 1.25)
  54      return (green > 70) & (green > red * 1.20) & (green > blue * 1.15)

  56  def detect_ball(image, color, matrix):
  57      mask = color_mask(image, color)
  58      rows, columns = np.nonzero(mask)
  59      if len(rows) < 4:
  60          return None
  61      # Median is robust to a few isolated colored pixels.
  62      pixel = np.array([np.median(rows), np.median(columns)], dtype=np.float64)
  63      world = pixel_to_table(pixel, matrix)
  64      world[:2] -= XY_BIAS
  65      return Detection(color, pixel, world, int(len(rows)))

  52      def act(self, observation):

  69          elif self.phase == "close":

复盘结论

语言被解析为两个顺序子任务；RGB检测建立初始场景记忆；状态机完成接近、下降、夹持、抬升、运输、稳定释放与退回；任务结束后以视觉误差复核。最终 GUI 是模块化 Language + Vision + Action 系统，不是大型端到端预训练 VLA。
